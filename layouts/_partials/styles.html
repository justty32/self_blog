{{- /* Keep cascade order explicit and publish one stylesheet at the stable URL. */ -}} {{- $paletteCSS := "" -}} {{- range $style := slice "grid" "paper" "deep" -}} {{- range $mode := slice "night" "day" -}} {{- $palette := index (index hugo.Data.appearance.styles $style) $mode -}} {{- $paletteCSS = printf "%s:root[data-style=%s][data-mode=%s]{" $paletteCSS $style $mode -}} {{- range $key := slice "bg" "surface" "text" "muted" "primary" "secondary" -}} {{- $color := index $palette $key -}} {{- if not (findRE `^#[0-9a-fA-F]{6}$` $color) -}} {{- errorf "data/appearance.toml: styles.%s.%s.%s must be a six-digit hex color" $style $mode $key -}} {{- end -}} {{- $paletteCSS = printf "%s--seed-%s:%s;" $paletteCSS $key $color -}} {{- end -}} {{- $paletteCSS = printf "%s}\n" $paletteCSS -}} {{- end -}} {{- end -}} {{- $palettes := resources.FromString "css/generated-palettes.css" $paletteCSS -}} {{- $styles := slice (resources.Get "css/base.css") $palettes (resources.Get "css/layout.css") (resources.Get "css/components.css") (resources.Get "css/prose.css") (resources.Get "css/scene.css") (resources.Get "css/themes.css") (resources.Get "css/preferences.css") -}} {{- return ($styles | resources.Concat "css/site.css") -}}
